using System;

namespace PowerLanguage.Strategy
{
	//
	// Summary:
	//     Allows access to strategy applied to specific traded symbol.
	public interface IPortfolioStrategy : IDisposable
	{
		//
		// Summary:
		//     Returns collection of the signals, which form the strategy.
		IStrategy[] Signals { get; }
		//
		// Summary:
		//     Allows/forbids order placement.
		bool Paused { get; set; }
		//
		// Summary:
		//     Returns the amount of capital in currency of the account that is taken by the
		//     strategy for marginal security (maximum loss).
		double RiskCapital { get; }
		//
		// Summary:
		//     Text message that is displayed in the "Custom Text" column of Portfolio Real-Time
		//     Window.
		string Status { set; }
		//
		// Summary:
		//     Returns/set the number of contracts to open market position. Allowed values:
		//     1) Less than "0" - not set. The number of contracts specified in code for each
		//     order is used. 2) "0" - default value. The number of contracts specified in Strategy
		//     Properties is used, no matter what number of contracts is specified for each
		//     order in code and in the Strategy Properties. 3) More than "0" - the set value
		//     is used for every entry, no matter what number of contracts is specified for
		//     each order in code and in the Strategy Properties.
		int EntryContracts { get; set; }
		//
		// Summary:
		//     Allows/forbids entry order placement.
		bool AllowEntries { get; set; }
		//
		// Summary:
		//     Allows/forbids long entry order placement.
		bool AllowEntriesLong { get; set; }
		//
		// Summary:
		//     Allows/forbids short entry order placement.
		bool AllowEntriesShort { get; set; }
		//
		// Summary:
		//     Allows/forbids exit order placement.
		bool AllowExits { get; set; }
		//
		// Summary:
		//     Allows/forbids long exit order placement.
		bool AllowExitsLong { get; set; }
		//
		// Summary:
		//     Allows/forbids short exit order placement.
		bool AllowExitsShort { get; set; }
		//
		// Summary:
		//     Allows access to variables of the strategy.
		IPortfolioData PortfolioData { get; }
		//
		// Summary:
		//     Returns index of the strategy in portfolio.
		int MyIndex { get; }

		//
		// Summary:
		//     Forced closing of open market position.
		void ForceClosePosition();
		//
		// Summary:
		//     Partial closing of open market position.
		void PartialClosePosition(bool isNextBar, int contracts);
	}
}

using ATCenterProxy.interop;

namespace PowerLanguage
{
	//
	// Summary:
	//     An interface describing the strategy.
	public interface IStrategy : IStudy
	{
		//
		// Summary:
		//     Read-only property. Returns strategy's MarketPosition.
		int PositionSide { get; }
		//
		// Summary:
		//     Read-only property. Returns 'Initial Capital' from the Strategy Properties window.
		double InitialCapital { get; }
		//
		// Summary:
		//     Read-only property. Returns 'Slippage' from the Strategy Properties window.
		double Slippage { get; }
		//
		// Summary:
		//     Read-only property. Returns 'Commission' from the Strategy Properties window.
		double Commission { get; }
		//
		// Summary:
		//     Read-only property. Returns a read-only collection of all strategy positions.
		IROList<IMarketPosition> Positions { get; }
		//
		// Summary:
		//     Read-only property. Returns strategy's current position.
		IMarketPosition CurrentPosition { get; }
		//
		// Summary:
		//     Read-only property. Returns an interface which provides Portfolio Perfomance.
		IPortfolioPerformance Portfolio { get; }
		//
		// Summary:
		//     Read-only property. Returns the total number of trades.
		int TotalTrades { get; }
		//
		// Summary:
		//     Read-only property. Returns the total number of bars during which winning trades
		//     were open.
		int TotalBarsWinTrades { get; }
		//
		// Summary:
		//     Read-only property. Returns the total number of bars during which losing trades
		//     were open.
		int TotalBarsLosTrades { get; }
		//
		// Summary:
		//     Read-only property. Returns the total number of bars during which even trades
		//     (without profit and loss) were open.
		int TotalBarsEvenTrades { get; }
		//
		// Summary:
		//     Returns a ratio of profitable trades to total number of trades expressed in percent
		//     ( 100*(NumWinTrades/TotalTrades)).
		double PercentProfit { get; }
		//
		// Summary:
		//     Read-only property. Returns total number or profitable trades.
		int NumWinTrades { get; }
		//
		// Summary:
		//     Read-only property. Returns total number or losing trades.
		int NumLosTrades { get; }
		//
		// Summary:
		//     Read-only property. Returns total number of trades that did not bring any profit
		//     or loss.
		int NumEvenTrades { get; }
		//
		// Summary:
		//     Read-only property. Returns maximum potential loss (drawdown) during entire trading
		//     period.
		double MaxDrawDown { get; }
		//
		// Summary:
		//     Read-only property. Returns maximum number of sequential unprofitable trades.
		int MaxLotsHeld { get; }
		//
		// Summary:
		//     Read-only property. Returns maximum number of sequential profitable trades.
		int MaxConsecWinners { get; }
		//
		// Summary:
		//     Read-only property. Returns maximum number of sequential unprofitable trades.
		int MaxConsecLosers { get; }
		//
		// Summary:
		//     Read-only property. Returns non-negative number with maximum profit of a single
		//     trade.
		double LargestWinTrade { get; }
		//
		// Summary:
		//     Read-only property. Returns non-positive number with maximum loss of a single
		//     trade.
		double LargestLosTrade { get; }
		//
		// Summary:
		//     Read-only property. Returns non-negative number that indicates total strategy
		//     profit.
		double GrossProfit { get; }
		//
		// Summary:
		//     Read-only property. Returns a non-positive number that indicates total strategy
		//     loss.
		double GrossLoss { get; }
		//
		// Summary:
		//     Read-only property. Returns the average length of winning trades in terms of
		//     bars.
		double AvgBarsWinTrade { get; }
		//
		// Summary:
		//     Read-only property. Returns the average length of losing trades in terms of bars.
		double AvgBarsLosTrade { get; }
		//
		// Summary:
		//     Read-only property. Returns the average length of even trades (without profit
		//     and loss) in terms of bars.
		double AvgBarsEvenTrade { get; }
		//
		// Summary:
		//     Read-only property. Returns 'Interest Rate' from the Strategy Properties window.
		double InterestRate { get; }
		//
		// Summary:
		//     Returns currency code from Strategy settings (Base Currency)
		MTPA_MCSymbolCurrency StrategyCurrencyCode { get; }
	}
}
